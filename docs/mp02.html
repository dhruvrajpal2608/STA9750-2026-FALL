get_eia_sep <- function(state, abbr){
    library(tidyverse)
    library(stringr)
    library(httr2)
    library(rvest)
    library(fs)
  
    state_formatted <- str_to_lower(state) |> str_replace_all("\\s", "")
    
    dir_name <- fs::path("data", "mp02")
    file_name <- fs::path(dir_name, state_formatted)
    
    fs::dir_create(dir_name, recurse=TRUE)
    
    if(!fs::file_exists(file_name)){
        BASE_URL <- "https://www.eia.gov"
        RESPONSE <- request(BASE_URL) |> 
          req_url_path("electricity", 
                       "state", 
                       state_formatted) |>
          req_headers(
            `User-Agent` = "Mozilla/5.0 (Macintosh; Intel Mac OS X 10.15; rv:149.0) Gecko/20100101 Firefox/149.0"
            ) |>   
          req_retry(
                max_tries = 4, 
          ) |>
          req_perform(file_name)
    
        resp_check_status(RESPONSE)
    }
    
    TABLE <- read_html(file_name) |> 
        html_element("table") |> 
        html_table() |>
        mutate(Item = str_to_lower(Item))
    
    if("U.S. rank" %in% colnames(TABLE)){
        TABLE <- TABLE |> rename(Rank = `U.S. rank`)
    }
    
    CO2_MWh <- TABLE |> 
        filter(Item == "carbon dioxide (lbs/mwh)") |>
        pull(Value) |> 
        str_replace_all(",", "") |>
        as.numeric()
    
    PRIMARY <- TABLE |> 
        filter(Item == "primary energy source") |> 
        pull(Rank)
    
    RATE <- TABLE |>
        filter(Item == "average retail price (cents/kwh)") |>
        pull(Value) |>
        as.numeric()
    
    GENERATION_MWh <- TABLE |>
        filter(Item == "net generation (megawatthours)") |>
        pull(Value) |>
        str_replace_all(",", "") |>
        as.numeric()
    
    data.frame(CO2_MWh               = CO2_MWh, 
               primary_source        = PRIMARY,
               electricity_price_MWh = RATE * 10, # / 100 cents to dollars &
               # * 1000 kWh to MWH 
               generation_MWh        = GENERATION_MWh, 
               state                 = state, 
               abbreviation          = abbr
    )
}

EIA_SEP_REPORT <- map2(state.name, state.abb, get_eia_sep) |> list_rbind()

load_ntd_energy <- function(){
  DATA_DIR <- fs::path("data", "mp02")
  fs::dir_create(DATA_DIR, recurse = TRUE)

  NTD_ENERGY_FILE <- fs::path(DATA_DIR, "2025_ntd_energy.xlsx")

  if(!fs::file_exists(NTD_ENERGY_FILE)){
    tryCatch({
      request("https://www.transit.dot.gov/") |>
        req_url_path_append("sites",
                            "fta.dot.gov",
                            "files",
                            "2026-09",
                            "2025%20Energy%20Consumption_260806.xlsx") |>
        req_headers(
          `User-Agent` = "Mozilla/5.0 (Macintosh; Intel Mac OS X 10.15; rv:149.0) Gecko/20100101 Firefox/149.0"
        ) |>
        req_retry(max_tries = 4) |>
        req_perform(path = NTD_ENERGY_FILE)
    }, error = function(e){
      unlink(NTD_ENERGY_FILE)
      stop("Download failed: ", conditionMessage(e), call. = FALSE)
    })
  }

  if(!fs::file_exists(NTD_ENERGY_FILE) || file.info(NTD_ENERGY_FILE)$size < 1e4){
    unlink(NTD_ENERGY_FILE)
    stop("Download failed or file is corrupted. The bad file was removed; please try again.")
  }

  to_numeric_fill_0 <- function(x){
    replace_na(as.numeric(x), 0)
  }

  read_excel(NTD_ENERGY_FILE) |>
    select(-any_of(c("Reporter Type",
                     "Reporting Module",
                     "Gallons of Other Fuel",
                     "Other Fuel Description"))) |>
    mutate(across(-any_of(c("NTD ID", "Agency Name", "Mode", "TOS")),
                  to_numeric_fill_0)) |>
    group_by(`NTD ID`, `Mode`, `Agency Name`) |>
    summarize(across(where(is.numeric), sum),
              .groups = "drop") |>
    mutate(ENERGY = rowSums(pick(-any_of(c("NTD ID", "Mode", "Agency Name"))))) |>
    filter(ENERGY > 0) |>
    select(-ENERGY)
}

load_ntd_service <- function(){
  library(httr2)
  library(tidyverse)
  library(fs)
  
  DATA_DIR <- fs::path("data", "mp02")
  fs::dir_create(DATA_DIR, recurse=TRUE)

  NTD_SERVICE_FILE <- fs::path(DATA_DIR, "2025_ntd_service.csv")

  if(!fs::file_exists(NTD_SERVICE_FILE)){
    library(httr2)
  
    request("https://data.transportation.gov/resource/4fir-qbim.csv") |>
      req_url_query(`$query`=paste(
        "SELECT  `_5_digit_ntd_id` AS `NTD_ID`", 
        "`max_agency` AS `AGENCY`",
        "`max_reporter_type` AS `reporter_type`",
        "`max_organization_type` AS `organization_type`",
        "`max_city` AS `city`",
        "`max_state` AS `state`",
        "`mode`",
        "`type_of_service`",
        "`max_primary_uza_code` as `uza_code`",
        "`max_primary_uza_name` as `uza_name`",
        "`max_primary_uza_area_sq_miles` as `uza_sq_miles`",
        "`max_primary_uza_population` as `uza_population`",
        "`max_service_area_sq_miles` as `service_area_sq_miles`",
        "`max_service_area_population` as `service_area_population`",
        "`questionable_record` as `flagged_likely_inaccurate`",
        "`sum_unlinked_passenger_trips_upt` as `total_upt`",
        "`sum_passenger_miles` WHERE caseless_one_of(`report_year`, '2025') limit 50000", 
        sep=",")) |>
      req_headers(
        `User-Agent` = "Mozilla/5.0 (Macintosh; Intel Mac OS X 10.15; rv:149.0) Gecko/20100101 Firefox/149.0") |> 
      req_perform(NTD_SERVICE_FILE)
  }
  
  if(file.info(NTD_SERVICE_FILE)$size < 1e4){
        cat("I was unable to download the NTD Service File. Please try again.\n")
        stop("Download failed. Please delete the corrupted file by running `unlink('data/mp02/2025_ntd_service.xlsx')` and try again.")
    }

  read_csv(NTD_SERVICE_FILE) |>
    mutate(flagged_likely_inaccurate = !is.na(flagged_likely_inaccurate))
}

  • Each row should include the EIA estimated CO2 emissions per MWh from the EIA_SEP_REPORT table.

  inner_join(agency_serv, by = "NTD_ID") |>

  • UPT Vs Passenger Mile: Urban bus and subway systems carrying high rider volumes over short distances excel on a per-UPT basis, whereas high-capacity commuter rail systems carrying passengers over tens of miles achieve their highest efficiencies on a per-passenger-mile basis.

WASHINGTON, D.C. — Today, researchers unveiled the winners of the inaugural GTA VI Green Transit Awards, an environmental benchmarking initiative honoring decarbonization leaders across the United States public transportation sector. Utilizing operational fleet reports from the Federal Transit Administration’s National Transit Database (NTD) paired with regional electricity emissions intensities from the U.S. Energy Information Administration (EIA), the awards evaluate how effectively agencies move passengers while minimizing their environmental footprint. Today’s honors recognize the transit systems that achieved the highest standard of climate efficiency, while highlighting the single greatest emissions challenge facing modern fleets.

  • Award 1: The Emerald Titan (Most Efficient Large Transit Agency) Category: Large Transit Authorities (\(\ge 10\text{M}\) Annual Unlinked Passenger Trips) Winner: Tri-County Metropolitan Transportation District of Oregon
    Winning Metric: 0.097 lbs \(\text{CO}_2\) per Passenger Mile Traveled (PMT)
    Metric Calculation: Emissions per passenger mile were determined by calculating gross agency emissions across all operated modes—summing direct fuel combustion factors with state-level grid electricity emissions (\(\text{MWh} \times \text{lbs CO}_2/\text{MWh}\))—and dividing that aggregate total by cumulative annual Passenger Miles Traveled (\(\text{PMT}\)).
    Benchmark Comparison: To summarize this achievement, the median peer transit agency in the United States generates approximately 1.08 lbs of \(\text{CO}_2\) per passenger mile, while the average gasoline-powered private commuter vehicle emits approximately 0.82 lbs of \(\text{CO}_2\) per passenger mile.Tri-County Metropolitan Transportation operates 83.2% cleaner than the national median transit peer, demonstrating the decarbonization potential of electrified, high-capacity urban rail.

  • Award 2: The Clean Commute Champion (Most Efficient Mid-to-Small Agency) Category: Regional & Municipal Transit Providers (\(100\text{K} - 10\text{M}\) Annual Trips) Winner: City of Seattle Winning Metric: 0.551 lbs \(\text{CO}_2\) per Unlinked Passenger Trip (UPT)
    Metric Calculation: The trip-level efficiency metric was computed by aggregating all direct combustion emissions and indirect propulsion electricity emissions, divided by the total count of unlinked boardings, \(\text{UPT}\)). This isolates first-and-last-mile trip cleanliness independently of route trip length.
